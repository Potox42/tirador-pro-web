// ================= Tienda =================
// SKU de los productos (así no dependemos de los ID).
const T360_SKU_BASICA = 'T360-BASICA';
const T360_SKU_PRO = 'T360-PRO';
const T360_SKU_SUBIR = 'T360-SUBIR-PRO';
const T360_URL_APP = 'https://potox42.github.io/app-armas/';

function t360_id_sku($sku) { return function_exists('wc_get_product_id_by_sku') ? (int) wc_get_product_id_by_sku($sku) : 0; }

function t360_carrito_tiene($sku) {
  if (!function_exists('WC') || !WC()->cart) return false;
  $id = t360_id_sku($sku);
  foreach (WC()->cart->get_cart() as $linea) if ((int) $linea['product_id'] === $id) return true;
  return false;
}

// Una licencia por pedido: al añadir un producto se vacía el carrito y se va directo al pago.
add_filter('woocommerce_add_to_cart_validation', function ($ok) {
  if ($ok && WC()->cart) WC()->cart->empty_cart();
  return $ok;
}, 1);
add_filter('woocommerce_add_to_cart_redirect', function () { return wc_get_checkout_url(); });

}, 10, 2);

// Pasar de Básica a Pro: hay que dar el código Básica.
add_action('woocommerce_after_order_notes', function ($checkout) {
  if (!t360_carrito_tiene(T360_SKU_SUBIR)) return;
  echo '<div class="t360-campo-subir"><h3>Tu código de Tirador 360 Básica</h3>';
  woocommerce_form_field('t360_codigo_basica', [
    'type' => 'text', 'required' => true, 'label' => 'Código Básica',
    'placeholder' => 'T360B-XXXX-XXXX-XXXX',
    'description' => 'Lo tienes en el correo de tu compra y en la app, en Perfil.',
  ], $checkout->get_value('t360_codigo_basica'));
  echo '</div>';
});

function t360_basica_valida($codigo) {
  if (!function_exists('lmfwc_get_license')) return 'No se ha podido comprobar el código. Escríbenos a rgg@ozono3.es.';
  $lic = lmfwc_get_license($codigo);
  if (!$lic || (int) $lic->getProductId() !== t360_id_sku(T360_SKU_BASICA) || !$lic->getOrderId()) {
    return 'Ese código no corresponde a ninguna licencia Básica. Revísalo (empieza por T360B-).';
  }
  $usado = wc_get_orders(['limit' => 1, 'status' => ['processing', 'completed'], 'return' => 'ids',
    'meta_query' => [['key' => '_t360_codigo_basica', 'value' => $codigo]]]);
  if ($usado) return 'Ese código Básica ya se ha usado para pasar a Pro.';
  return '';
}

add_action('woocommerce_checkout_process', function () {
  if (!t360_carrito_tiene(T360_SKU_SUBIR)) return;
  $codigo = strtoupper(trim(sanitize_text_field(wp_unslash($_POST['t360_codigo_basica'] ?? ''))));
  if ($codigo === '') { wc_add_notice('Escribe tu código de Tirador 360 Básica.', 'error'); return; }
  $error = t360_basica_valida($codigo);
  if ($error) wc_add_notice($error, 'error');
});

add_action('woocommerce_checkout_create_order', function ($order) {
  if (!empty($_POST['t360_codigo_basica'])) {
    $order->update_meta_data('_t360_codigo_basica', strtoupper(trim(sanitize_text_field(wp_unslash($_POST['t360_codigo_basica'])))));
  }
});

// Al completarse el paso a Pro, la licencia Básica queda desactivada (el cliente usa el código Pro nuevo).
add_action('woocommerce_order_status_completed', function ($order_id) {
  $order = wc_get_order($order_id);
  $codigo = $order ? $order->get_meta('_t360_codigo_basica') : '';
  if ($codigo && function_exists('lmfwc_update_license')) {
    try { lmfwc_update_license($codigo, ['status' => 4]); $order->add_order_note('Licencia Básica ' . $codigo . ' desactivada por el paso a Pro.'); }
    catch (\Throwable $e) { $order->add_order_note('No se pudo desactivar la licencia Básica ' . $codigo . ': ' . $e->getMessage()); }
  }
});

// ================= Recomendados =================
// Cada compra de Básica o Pro genera un código AMIGO-XXXXX: 10 % de descuento en Básica o Pro,
// válido para 3 compras de otras personas. Se envía en el correo de compra y la app lo comparte.
function t360_crear_codigo_amigo($order) {
  // El correo y el cambio de estado llegan con copias distintas del pedido: se mira el guardado.
  static $hechos = [];
  $id = $order->get_id();
  if (isset($hechos[$id])) return $hechos[$id];
  $guardado = wc_get_order($id);
  $ya = $guardado ? $guardado->get_meta('_t360_codigo_amigo') : '';
  if ($ya) return $hechos[$id] = $ya;
  $order = $guardado ?: $order;
  $ids = [t360_id_sku(T360_SKU_BASICA), t360_id_sku(T360_SKU_PRO)];
  $compra = false;
  foreach ($order->get_items() as $item) if (in_array((int) $item->get_product_id(), $ids, true)) $compra = true;
  if (!$compra) return $hechos[$id] = '';
  $letras = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  do {
    $codigo = 'AMIGO-';
    for ($i = 0; $i < 5; $i++) $codigo .= $letras[random_int(0, strlen($letras) - 1)];
  } while (wc_get_coupon_id_by_code($codigo));
  $cupon = new WC_Coupon();
  $cupon->set_code($codigo);
  $cupon->set_discount_type('percent');
  $cupon->set_amount(10);
  $cupon->set_usage_limit(3);
  $cupon->set_usage_limit_per_user(1);
  $cupon->set_individual_use(true);
  $cupon->set_product_ids(array_filter($ids));
  $cupon->set_email_restrictions([]);
  $cupon->set_description('Recomendado por el pedido #' . $order->get_id() . ' (' . $order->get_billing_email() . ')');
  $cupon->update_meta_data('_t360_pedido_origen', $order->get_id());
  $cupon->update_meta_data('_t360_email_origen', strtolower($order->get_billing_email()));
  $cupon->save();
  $order->update_meta_data('_t360_codigo_amigo', $codigo);
  $order->save();
  $order->add_order_note('Código de recomendado creado: ' . $codigo);
  return $hechos[$id] = $codigo;
}
// Prioridad 5: antes de que salga el correo de pedido completado.
add_action('woocommerce_order_status_completed', function ($order_id) {
  $order = wc_get_order($order_id);
  if ($order) t360_crear_codigo_amigo($order);
}, 5);

// Nadie puede usar su propio código.
add_filter('woocommerce_coupon_is_valid', function ($valido, $cupon) {
  $origen = $cupon->get_meta('_t360_email_origen');
  if (!$valido || !$origen || !WC()->customer) return $valido;
  $email = strtolower((string) WC()->customer->get_billing_email());
  if ($email && $email === $origen) throw new Exception('No puedes usar tu propio código de recomendado.');
  return $valido;
}, 10, 2);

// Enlace con ?amigo=CODIGO: se guarda y se aplica solo al pagar.
add_action('template_redirect', function () {
  if (empty($_GET['amigo']) || !function_exists('WC') || !WC()->session) return;
  $codigo = strtoupper(sanitize_text_field(wp_unslash($_GET['amigo'])));
  if (!preg_match('/^AMIGO-[A-Z0-9]{5}$/', $codigo)) return;
  if (!WC()->session->has_session()) WC()->session->set_customer_session_cookie(true);
  WC()->session->set('t360_amigo', $codigo);
});
add_action('woocommerce_before_checkout_form', function () {
  if (!WC()->session || !WC()->cart) return;
  $codigo = WC()->session->get('t360_amigo');
  if (!$codigo || WC()->cart->has_discount($codigo) || t360_carrito_tiene(T360_SKU_SUBIR)) return;
  if (WC()->cart->apply_coupon($codigo)) WC()->session->set('t360_amigo', null);
}, 5);
add_filter('woocommerce_coupon_message', function ($mensaje, $codigo_msg, $cupon) {
  return ($cupon && strpos($cupon->get_code(), 'amigo-') === 0) ? 'Descuento de recomendado aplicado: 10 %.' : $mensaje;
}, 10, 3);
