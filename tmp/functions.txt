// ================= Tienda =================
// SKU de los productos (así no dependemos de los ID).
const T360_SKU_BASICA = 'T360-BASICA';
const T360_SKU_PRO = 'T360-PRO';
const T360_SKU_SUBIR = 'T360-SUBIR-PRO';
const T360_URL_APP = 'https://potox42.github.io/app-armas/';

function t360_id_sku($sku) { return function_exists('wc_get_product_id_by_sku') ? (int) wc_get_product_id_by_sku($sku) : 0; }

function t360_carrito_tiene($sku) {
  if (!function_exists('WC') || !WC()->cart) return false;
  $id = t360_id_sku($sku);
  foreach (WC()->cart->get_cart() as $linea) if ((int) $linea['product_id'] === $id) return true;
  return false;
}

// Una licencia por pedido: al añadir un producto se vacía el carrito y se va directo al pago.
add_filter('woocommerce_add_to_cart_validation', function ($ok) {
  if ($ok && WC()->cart) WC()->cart->empty_cart();
  return $ok;
}, 1);
add_filter('woocommerce_add_to_cart_redirect', function () { return wc_get_checkout_url(); });
add_filter('wc_add_to_cart_message_html', '__return_empty_string');

// Los productos son virtuales: el pedido pagado pasa solo a «Completado» y así se envía el código.
add_filter('woocommerce_order_item_needs_processing', function ($necesita, $producto) {
  return ($producto && $producto->is_virtual()) ? false : $necesita;
}, 10, 2);

// Producto digital: en el pago solo pedimos lo necesario (sin dirección ni notas de entrega).
add_filter('woocommerce_enable_order_notes_field', '__return_false');
add_filter('woocommerce_checkout_fields', function ($campos) {
  foreach (['billing_company', 'billing_address_1', 'billing_address_2', 'billing_city', 'billing_state', 'billing_postcode'] as $k) unset($campos['billing'][$k]);
  if (isset($campos['billing']['billing_phone'])) $campos['billing']['billing_phone']['required'] = false;
  return $campos;
});

// Casilla de condiciones con la renuncia al desistimiento (art. 103 m TRLGDCU).
add_filter('woocommerce_get_terms_and_conditions_checkbox_text', function () {
  return 'He leído y acepto los [terms]. Solicito recibir mi código de licencia ahora mismo y entiendo que, una vez recibido, pierdo el derecho de desistimiento.';
});

// Pasar de Básica a Pro: hay que dar el código Básica.
add_action('woocommerce_after_order_notes', function ($checkout) {
  if (!t360_carrito_tiene(T360_SKU_SUBIR)) return;
  echo '<div class="t360-campo-subir"><h3>Tu código de Tirador 360 Básica</h3>';
  woocommerce_form_field('t360_codigo_basica', [
    'type' => 'text', 'required' => true, 'label' => 'Código Básica',
    'placeholder' => 'T360B-XXXX-XXXX-XXXX',
    'description' => 'Lo tienes en el correo de tu compra y en la app, en Perfil.',
  ], $checkout->get_value('t360_codigo_basica'));
  echo '</div>';
});

function t360_basica_valida($codigo) {
  if (!function_exists('lmfwc_get_license')) return 'No se ha podido comprobar el código. Escríbenos a rgg@ozono3.es.';
  $lic = lmfwc_get_license($codigo);
  if (!$lic || (int) $lic->getProductId() !== t360_id_sku(T360_SKU_BASICA) || !$lic->getOrderId()) {
    return 'Ese código no corresponde a ninguna licencia Básica. Revísalo (empieza por T360B-).';
  }
  $usado = wc_get_orders(['limit' => 1, 'status' => ['processing', 'completed'], 'return' => 'ids',
    'meta_query' => [['key' => '_t360_codigo_basica', 'value' => $codigo]]]);
  if ($usado) return 'Ese código Básica ya se ha usado para pasar a Pro.';
  return '';
}

add_action('woocommerce_checkout_process', function () {
  if (!t360_carrito_tiene(T360_SKU_SUBIR)) return;
  $codigo = strtoupper(trim(sanitize_text_field(wp_unslash($_POST['t360_codigo_basica'] ?? ''))));
  if ($codigo === '') { wc_add_notice('Escribe tu código de Tirador 360 Básica.', 'error'); return; }
  $error = t360_basica_valida($codigo);
  if ($error) wc_add_notice($error, 'error');
});

add_action('woocommerce_checkout_create_order', function ($order) {
  if (!empty($_POST['t360_codigo_basica'])) {
    $order->update_meta_data('_t360_codigo_basica', strtoupper(trim(sanitize_text_field(wp_unslash($_POST['t360_codigo_basica'])))));
  }
});

// Al completarse el paso a Pro, la licencia Básica queda desactivada (el cliente usa el código Pro nuevo).
add_action('woocommerce_order_status_completed', function ($order_id) {
  $order = wc_get_order($order_id);
  $codigo = $order ? $order->get_meta('_t360_codigo_basica') : '';
  if ($codigo && function_exists('lmfwc_update_license')) {
    try { lmfwc_update_license($codigo, ['status' => 4]); $order->add_order_note('Licencia Básica ' . $codigo . ' desactivada por el paso a Pro.'); }
    catch (\Throwable $e) { $order->add_order_note('No se pudo desactivar la licencia Básica ' . $codigo . ': ' . $e->getMessage()); }
  }
});

add_action('woocommerce_admin_order_data_after_billing_address', function ($order) {
  $c = $order->get_meta('_t360_codigo_basica');
  if ($c) echo '<p><strong>Código Básica (paso a Pro):</strong> ' . esc_html($c) . '</p>';
});

  $pasos = [
    'Abre en tu móvil <a href="' . esc_url(T360_URL_APP) . '">' . esc_html(T360_URL_APP) . '</a> (en iPhone con Safari; en Android con Chrome).',

    'Abre Tirador 360 desde el icono, ve a Perfil y escribe tu código de licencia (lo tienes más abajo en este correo).',
  ];
  if ($texto_plano) {
    echo "CÓMO EMPEZAR\n";
    foreach ($pasos as $i => $p) echo ($i + 1) . '. ' . wp_strip_all_tags($p) . "\n";
    echo "\nConfirmamos que solicitaste recibir el código en el momento de la compra y que, por tratarse de contenido digital ya entregado, no se aplica el derecho de desistimiento.\n\n";
    return;
  }
  echo '<h2>Cómo empezar</h2><ol>';
  foreach ($pasos as $p) echo '<li style="margin-bottom:6px">' . $p . '</li>';
  echo '</ol><p>Tu licencia funciona en un móvil. Si cambias de teléfono, haz una copia de seguridad, pulsa «Liberar licencia» en Perfil y activa el mismo código en el móvil nuevo.</p>';
  echo '<p style="font-size:12px;color:#666">Confirmamos que solicitaste recibir el código en el momento de la compra y que, por tratarse de contenido digital ya entregado, no se aplica el derecho de desistimiento.</p>';
}, 10, 4);
